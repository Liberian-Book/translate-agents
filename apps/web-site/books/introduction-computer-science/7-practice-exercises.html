7 .

8 .

9 .
Interpret the following program snippet. What will the function return? 
int main()
{
   int x = 1;
   int y = 7;

   if (x == 1 && y > 10)
   {
       return 0;
   }
   else if (x < 0 || y == x)
   {
       return 1;
   }
   else if (x != 0 && y == x)
   {
       return 2;
   }
   else if (x > 0 && (x != y || y > x))
   {
       return 3;
   }
   else
   {
       return 4;
   }
}
